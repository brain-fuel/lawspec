law `left inverse`
  (f :: b -> a)
  (g :: a -> b)
requires
  Eq a
is
  definition is
    `for all` (x :: a) .
      (f . g) x = x
  end

  description is
    "applying {g} to a value and then applying {f} yields the original value"
  end

  rationale is
    "{g} must preserve enough information for {f} to recover the original value"
  end
end

  (a :: x -> y)
  (b :: y -> x)

  description is
    "applying {a} to a value and then applying {b} yields the original value"
  end

basically, we need to figure a good UX and make this work with javascript | fastcheck; typescipt | fastcheck; go | rapid; Haskell | Hedgehog + HSpec; Kotlin | Kotest Property
